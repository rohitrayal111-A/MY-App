  <script>
    // State storage keys
    const STORAGE_KEY = 'life_os_data_v1';
    let db = JSON.parse(localStorage.getItem(STORAGE_KEY)) || {
      tasks: [],
      logs: [],
      habits: {},
      goals: [],
      reviews: {}
    };

    let currentDate = new Date().toISOString().split('T')[0];
    document.getElementById('selected-date').value = currentDate;
    let currentRating = 0;
    let chartInstance = null;

    function saveState() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(db));
      renderAll();
    }

    function switchTab(tabId, el) {
      document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
      document.getElementById(tabId).classList.add('active');
      el.classList.add('active');
      if (tabId === 'tab-analytics') renderChart();
    }

    function loadDateData() {
      currentDate = document.getElementById('selected-date').value;
      renderAll();
    }

    // Task Management
    function addTask(e) {
      e.preventDefault();
      const title = document.getElementById('task-title').value.trim();
      const time = document.getElementById('task-time').value;
      const cat = document.getElementById('task-cat').value;
      db.tasks.push({ id: Date.now(), title, time, cat, date: currentDate, done: false });
      document.getElementById('task-title').value = '';
      saveState();
    }

    function toggleTask(id) {
      db.tasks = db.tasks.map(t => t.id === id ? { ...t, done: !t.done } : t);
      saveState();
    }

    function deleteTask(id) {
      db.tasks = db.tasks.filter(t => t.id !== id);
      saveState();
    }

    // Activity Log
    function addLog(e) {
      e.preventDefault();
      const text = document.getElementById('log-text').value.trim();
      const now = new Date();
      const timeStr = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      db.logs.push({ id: Date.now(), text, time: timeStr, date: currentDate });
      document.getElementById('log-text').value = '';
      saveState();
    }

    function deleteLog(id) {
      db.logs = db.logs.filter(l => l.id !== id);
      saveState();
    }

    // Wellness & Habits
    function getHabitRecord() {
      if (!db.habits[currentDate]) {
        db.habits[currentDate] = { water: 0, running: 0, yoga: 0, meditation: 0 };
      }
      return db.habits[currentDate];
    }

    function updateWater(val) {
      const h = getHabitRecord();
      h.water = Math.max(0, h.water + val);
      saveState();
    }
    function updateRunning(val) {
      const h = getHabitRecord();
      h.running = Math.max(0, h.running + val);
      saveState();
    }
    function updateYoga(val) {
      const h = getHabitRecord();
      h.yoga = Math.max(0, h.yoga + val);
      saveState();
    }
    function updateMeditation(val) {
      const h = getHabitRecord();
      h.meditation = Math.max(0, h.meditation + val);
      saveState();
    }

    // Goals
    function addGoal(e) {
      e.preventDefault();
      const title = document.getElementById('goal-title').value.trim();
      const deadline = document.getElementById('goal-date').value;
      const tier = document.getElementById('goal-tier').value;
      db.goals.push({ id: Date.now(), title, deadline, tier, progress: 0 });
      document.getElementById('goal-title').value = '';
      saveState();
    }

    function updateGoalProgress(id, val) {
      db.goals = db.goals.map(g => g.id === id ? { ...g, progress: val } : g);
      saveState();
    }

    function deleteGoal(id) {
      db.goals = db.goals.filter(g => g.id !== id);
      saveState();
    }

    // Review
    function setStars(val) {
      currentRating = val;
      const stars = document.querySelectorAll('#star-box .star');
      stars.forEach((s, idx) => {
        s.classList.toggle('selected', idx < val);
      });
    }

    function saveReview() {
      const wins = document.getElementById('rev-wins').value;
      const learn = document.getElementById('rev-learn').value;
      const hurdles = document.getElementById('rev-hurdles').value;
      db.reviews[currentDate] = { rating: currentRating, wins, learn, hurdles };
      saveState();
      alert("Review saved for " + currentDate);
    }

    // Renders
    function renderAll() {
      // Tasks
      const dayTasks = db.tasks.filter(t => t.date === currentDate).sort((a,b) => a.time.localeCompare(b.time));
      const taskList = document.getElementById('task-list');
      taskList.innerHTML = dayTasks.length === 0 ? '<li style="color:var(--muted); font-size:0.85rem;">No tasks scheduled for this date.</li>' : '';
      dayTasks.forEach(t => {
        const li = document.createElement('li');
        li.className = `item ${t.done ? 'completed' : ''}`;
        let tagClass = t.cat === 'Work' ? 'tag-work' : t.cat === 'Ops' ? 'tag-ops' : 'tag-growth';
        li.innerHTML = `
          <div class="item-left">
            <input type="checkbox" ${t.done ? 'checked' : ''} onchange="toggleTask(${t.id})">
            <div>
              <div class="item-text" style="font-size:0.9rem;">${t.title}</div>
              <div style="font-size:0.75rem; color:var(--muted); margin-top:2px;">
                <span>⏰ ${t.time}</span> • <span class="tag ${tagClass}">${t.cat}</span>
              </div>
            </div>
          </div>
          <button class="delete-btn" onclick="deleteTask(${t.id})">✕</button>
        `;
        taskList.appendChild(li);
      });

      // Logs
      const dayLogs = db.logs.filter(l => l.date === currentDate);
      const logList = document.getElementById('log-list');
      logList.innerHTML = dayLogs.length === 0 ?
